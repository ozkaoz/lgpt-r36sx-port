   101	void TreeFrogEventManager::PushPad(GUIEventPadButtonType button, bool pressed) {
   102	    long when = System::GetInstance()->GetClock();
   103	
   104	    if (g_qcount >= (sizeof(g_queue) / sizeof(g_queue[0]))) {
   105	        /* Drop the oldest event rather than corrupting the queue. */
   106	        g_qread = (g_qread + 1) % (sizeof(g_queue) / sizeof(g_queue[0]));
   107	        --g_qcount;
   108	        logEvent("drop-oldest", button, pressed, g_qcount);
   109	    }
   110	
   111	    g_queue[g_qwrite].button = button;
   112	    g_queue[g_qwrite].pressed = pressed;
   113	    g_queue[g_qwrite].when = when;
   114	    g_qwrite = (g_qwrite + 1) % (sizeof(g_queue) / sizeof(g_queue[0]));
   115	    ++g_qcount;
   116	
   117	#if TREEFROG_EVENT_DEBUG_ALL
   118	    logEvent("queued", button, pressed, g_qcount);
   119	#else
   120	    if (button == EPBT_START || button == EPBT_SELECT) {
   121	        logEvent("queued", button, pressed, g_qcount);
   122	    }
   123	#endif
   124	}

   126	void TreeFrogEventManager::Flush() {
   127	    GUIWindow *window = Application::GetInstance()->GetWindow();
   128	    if (!window) {
   129	        ClearQueue();
   130	        return;
   131	    }
   132	
   133	    unsigned safety = 0;
   134	    while (g_qcount > 0 && safety++ < 64) {
   135	        QueuedPadEvent ev = g_queue[g_qread];
   136	        g_qread = (g_qread + 1) % (sizeof(g_queue) / sizeof(g_queue[0]));
   137	        --g_qcount;
   138	
   139	#if TREEFROG_EVENT_DEBUG_ALL
   140	        logEvent("dispatch.enter", ev.button, ev.pressed, g_qcount);
   141	#else
   142	        if (ev.button == EPBT_START || ev.button == EPBT_SELECT) {
   143	            logEvent("dispatch.enter", ev.button, ev.pressed, g_qcount);
   144	        }
   145	#endif
   146	
   147	        GUIEvent e((long)ev.button, ev.pressed ? ET_PADBUTTONDOWN : ET_PADBUTTONUP, ev.when);
   148	        window->DispatchEvent(e);
   149	
   150	#if TREEFROG_EVENT_DEBUG_ALL
   151	        logEvent("dispatch.leave", ev.button, ev.pressed, g_qcount);
   152	#else
   153	        if (ev.button == EPBT_START || ev.button == EPBT_SELECT) {
   154	            logEvent("dispatch.leave", ev.button, ev.pressed, g_qcount);
   155	        }
   156	#endif
   157	    }
   158	}
